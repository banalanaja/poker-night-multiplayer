cmake_minimum_required(VERSION 3.25)
project(PokerNightBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(PokerNightBridge SHARED Bridge.cpp)
target_compile_definitions(PokerNightBridge PRIVATE PNM_BUILD)
target_compile_options(PokerNightBridge PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4> $<$<CXX_COMPILER_ID:MSVC>:/WX>)
add_executable(BridgeTests BridgeTests.cpp)
enable_testing()
add_test(NAME BridgeQueue COMMAND BridgeTests)
